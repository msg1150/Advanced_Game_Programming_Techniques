cmake_minimum_required(VERSION 3.20)
project(DXFramework_v018 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(DX_BUILD_APP "Build Editor and Player" ON)
option(DX_BUILD_LEGACY_DEMO "Build v0.17 demonstration workbench" OFF)
option(DX_ENABLE_GPU_PICKING "Legacy GPU picking" ON)
option(DX_ENABLE_SCENE_PERSISTENCE "Legacy collider documents" ON)
option(DX_ENABLE_TERRAIN_QUERY "Terrain queries" ON)
option(DX_ENABLE_TERRAIN_PLACEMENT "Legacy terrain placement" ${DX_ENABLE_TERRAIN_QUERY})
option(DX_ENABLE_WORLD_TERRAIN "Project terrain module" ON)
option(DX_ENABLE_ENGINE_RUNTIME "Legacy runtime feature/component suite" ON)
option(DX_ENABLE_ENGINE_GRAPHICS "Mesh/texture/shader backend" ON)
option(DX_ENABLE_ENGINE_COLLISION "Object collision bridge" ON)
option(DX_ENABLE_ENGINE_GAMEPLAY "Movement components" ON)
if(NOT DX_ENABLE_TERRAIN_QUERY)
    set(DX_ENABLE_TERRAIN_PLACEMENT OFF)
endif()
if(NOT DX_ENABLE_ENGINE_RUNTIME)
    set(DX_ENABLE_ENGINE_GRAPHICS OFF)
    set(DX_ENABLE_ENGINE_COLLISION OFF)
    set(DX_ENABLE_ENGINE_GAMEPLAY OFF)
endif()
set(DX_FEATURES GPU_PICKING SCENE_PERSISTENCE TERRAIN_QUERY TERRAIN_PLACEMENT WORLD_TERRAIN ENGINE_RUNTIME ENGINE_GRAPHICS ENGINE_COLLISION ENGINE_GAMEPLAY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/Tools/TargetSources.json" DX_TARGETS)
function(dx_sources role output)
    string(JSON n LENGTH "${DX_TARGETS}" ${role})
    math(EXPR last "${n}-1")
    set(result)
    foreach(i RANGE 0 ${last})
        string(JSON path GET "${DX_TARGETS}" ${role} ${i})
        if(path MATCHES "TileArchive.cpp" AND NOT DX_ENABLE_WORLD_TERRAIN AND NOT DX_ENABLE_TERRAIN_QUERY)
            continue()
        elseif(path MATCHES "/Engine/Graphics/" AND NOT DX_ENABLE_ENGINE_GRAPHICS)
            continue()
        elseif(path MATCHES "/Engine/Collision/" AND NOT DX_ENABLE_ENGINE_COLLISION)
            continue()
        elseif(path MATCHES "/Terrain/Query/" AND NOT DX_ENABLE_TERRAIN_QUERY)
            continue()
        elseif(path MATCHES "/(DiskTerrainStreaming|QuadTreeLOD|QuadTreeLODGeomorph|TriplanarTerrain)/" AND NOT path MATCHES "TileArchive.cpp" AND NOT DX_ENABLE_WORLD_TERRAIN)
            continue()
        endif()
        list(APPEND result "${CMAKE_CURRENT_SOURCE_DIR}/${path}")
    endforeach()
    set(${output} ${result} PARENT_SCOPE)
endfunction()
if(DX_BUILD_APP)
    if(NOT WIN32)
        message(FATAL_ERROR "Editor/Player require Win32; use DX_BUILD_APP=OFF for portable tests")
    endif()
    dx_sources(runtime RUNTIME_SOURCES)
    add_library(DXRuntime STATIC ${RUNTIME_SOURCES})
    target_include_directories(DXRuntime PUBLIC Source)
    target_compile_definitions(DXRuntime PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    foreach(feature IN LISTS DX_FEATURES)
        target_compile_definitions(DXRuntime PUBLIC DX_ENABLE_${feature}=$<BOOL:${DX_ENABLE_${feature}}>)
    endforeach()
    target_link_libraries(DXRuntime PUBLIC d3d11 dxgi d3dcompiler ole32 windowscodecs shell32)
    dx_sources(editor EDITOR_SOURCES)
    add_executable(DXEditor WIN32 ${EDITOR_SOURCES} Source/Platform/app.manifest)
    add_executable(DXPlayer WIN32 Source/Player/Main.cpp Source/Platform/app.manifest)
    target_link_libraries(DXEditor PRIVATE DXRuntime d2d1 dwrite comdlg32 uuid gdi32)
    target_link_libraries(DXPlayer PRIVATE DXRuntime)
    foreach(target DXRuntime DXEditor DXPlayer)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
        endif()
    endforeach()
    foreach(target DXEditor DXPlayer)
        add_custom_command(TARGET ${target} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/Assets" "$<TARGET_FILE_DIR:${target}>/Assets"
            COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/Shaders" "$<TARGET_FILE_DIR:${target}>/Shaders"
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/Starter.dxproject" "$<TARGET_FILE_DIR:${target}>/Starter.dxproject")
    endforeach()
    if(DX_BUILD_LEGACY_DEMO)
        if(NOT DX_ENABLE_WORLD_TERRAIN)
            message(FATAL_ERROR "The preserved legacy demo requires terrain; Editor/Player do not")
        endif()
        dx_sources(legacy LEGACY_SOURCES)
        # Legacy target preserves its v0.17 feature source exclusions.
        if(NOT DX_ENABLE_GPU_PICKING)
            list(FILTER LEGACY_SOURCES EXCLUDE REGEX "/GPUPicking/")
        endif()
        if(NOT DX_ENABLE_SCENE_PERSISTENCE)
            list(FILTER LEGACY_SOURCES EXCLUDE REGEX "/(ScenePersistence/|Scene/(Runtime|Serialization)/)")
        endif()
        if(NOT DX_ENABLE_TERRAIN_PLACEMENT)
            list(FILTER LEGACY_SOURCES EXCLUDE REGEX "/TerrainPlacement/")
        endif()
        add_executable(DXLegacyDemo WIN32 ${LEGACY_SOURCES} Source/Platform/app.manifest)
        target_link_libraries(DXLegacyDemo PRIVATE DXRuntime d2d1 dwrite comdlg32 uuid)
        if(MSVC)
            target_compile_options(DXLegacyDemo PRIVATE /W4 /utf-8)
        endif()
    endif()
endif()

    add_subdirectory(Tests)
endif()
