function(dx_test name)
    add_executable(${name} ${name}.cpp ${ARGN})
    target_include_directories(${name} PRIVATE ${ROOT}/Source)
    if(WIN32)
        target_compile_definitions(${name} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    endif()
    if(MSVC)
        target_compile_options(${name} PRIVATE /W4 /utf-8 /UNDEBUG /FI${ROOT}/Tests/TestDiagnostics.h)
    else()
        target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
    endif()
    if(name STREQUAL "test_project_unicode")
        add_test(NAME ${name} COMMAND ${name} ${CMAKE_CURRENT_BINARY_DIR}/unicode-data)
    elseif(name STREQUAL "test_process_unicode")
        add_test(NAME ${name} COMMAND ${name} $<TARGET_FILE:unicode_args_helper> ${CMAKE_CURRENT_BINARY_DIR}/unicode-data)
    elseif(name STREQUAL "test_unicode_runtime")
        add_test(NAME ${name} COMMAND ${name} ${ROOT} ${CMAKE_CURRENT_BINARY_DIR}/unicode-data)
    elseif(name STREQUAL "test_tile_archive")
        add_test(NAME ${name} COMMAND ${name} ${ROOT}/Assets/TerrainTiles)
    elseif(name STREQUAL "test_gpu_picking")
        add_test(NAME ${name} COMMAND ${name} ${ROOT}/Shaders)
    else()
        add_test(NAME ${name} COMMAND ${name})
    endif()
endfunction()
