cmake_minimum_required(VERSION 3.20)
project(DXFramework_v017 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(DX_BUILD_APP "Build Windows terrain runtime" ON)
option(DX_ENABLE_GPU_PICKING "Attach GPU picking module" ON)
option(DX_ENABLE_SCENE_PERSISTENCE "Attach collider scene file module" ON)
option(DX_ENABLE_TERRAIN_QUERY "Build portable terrain query module" ON)
option(DX_ENABLE_TERRAIN_PLACEMENT "Attach terrain placement tool" ${DX_ENABLE_TERRAIN_QUERY})
if(DX_ENABLE_TERRAIN_PLACEMENT AND NOT DX_ENABLE_TERRAIN_QUERY)
    message(STATUS "Terrain placement omitted: query provider is disabled")
    set(DX_ENABLE_TERRAIN_PLACEMENT OFF)
endif()
option(DX_ENABLE_ENGINE_RUNTIME "Build GameObject/Scene runtime" ON)
option(DX_ENABLE_ENGINE_GRAPHICS "Attach object graphics/resources" ON)
option(DX_ENABLE_ENGINE_COLLISION "Attach object collision adapter" ON)
option(DX_ENABLE_ENGINE_GAMEPLAY "Attach fixed-step movement" ON)
if(NOT DX_ENABLE_ENGINE_RUNTIME)
    set(DX_ENABLE_ENGINE_GRAPHICS OFF)
    set(DX_ENABLE_ENGINE_COLLISION OFF)
    set(DX_ENABLE_ENGINE_GAMEPLAY OFF)
endif()
if(DX_BUILD_APP)
    if(NOT WIN32)
        message(FATAL_ERROR "The runtime uses Win32/D3D11; use -DDX_BUILD_APP=OFF for portable tests")
    endif()
    file(GLOB_RECURSE DX_SOURCES CONFIGURE_DEPENDS Source/*.cpp)
    list(FILTER DX_SOURCES EXCLUDE REGEX "/(Features/(GPUPicking|ScenePersistence|TerrainPlacement|EngineRuntime)|Engine/|Scene/|Terrain/Query/|Platform/SceneFileDialogs)")
    add_executable(DXFramework WIN32 ${DX_SOURCES})
    target_include_directories(DXFramework PRIVATE Source)
    target_compile_definitions(DXFramework PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    foreach(feature GPU_PICKING SCENE_PERSISTENCE TERRAIN_QUERY TERRAIN_PLACEMENT ENGINE_RUNTIME ENGINE_GRAPHICS ENGINE_COLLISION ENGINE_GAMEPLAY)
        target_compile_definitions(DXFramework PRIVATE DX_ENABLE_${feature}=$<BOOL:${DX_ENABLE_${feature}}>)
    endforeach()
    if(DX_ENABLE_GPU_PICKING)
        target_sources(DXFramework PRIVATE Source/Features/GPUPicking/GPUPickingSystem.cpp)
    endif()
    if(DX_ENABLE_SCENE_PERSISTENCE)
        target_sources(DXFramework PRIVATE Source/Scene/Serialization/ColliderSceneDocument.cpp Source/Scene/Runtime/ColliderSceneAdapter.cpp Source/Features/ScenePersistence/ScenePersistenceFeature.cpp  )
    endif()
    if(DX_ENABLE_TERRAIN_QUERY)
        target_sources(DXFramework PRIVATE Source/Terrain/Query/TerrainQuery.cpp Source/Terrain/Query/DiskTerrainQuerySource.cpp)
    endif()
    if(DX_ENABLE_TERRAIN_PLACEMENT)
        target_sources(DXFramework PRIVATE Source/Features/TerrainPlacement/TerrainPlacementFeature.cpp)
    endif()
    if(DX_ENABLE_SCENE_PERSISTENCE OR DX_ENABLE_ENGINE_RUNTIME)
        target_sources(DXFramework PRIVATE Source/Platform/SceneFileDialogs.cpp)
    endif()
    if(DX_ENABLE_ENGINE_RUNTIME)
        target_sources(DXFramework PRIVATE Source/Engine/Core/Scene.cpp Source/Engine/Serialization/SceneSerializer.cpp Source/Features/EngineRuntime/EngineRuntimeFeature.cpp)
        if(DX_ENABLE_ENGINE_COLLISION)
            target_sources(DXFramework PRIVATE Source/Engine/Collision/CollisionBody.cpp)
        endif()
        if(DX_ENABLE_ENGINE_GRAPHICS)
            target_sources(DXFramework PRIVATE Source/Engine/Graphics/GraphicsResources.cpp)
        endif()
    endif()
    if(MSVC)
        target_compile_options(DXFramework PRIVATE /W4 /permissive- /utf-8)
    endif()
    target_link_libraries(DXFramework PRIVATE d3d11 dxgi d3dcompiler ole32 windowscodecs d2d1 dwrite comdlg32)
    add_custom_command(TARGET DXFramework POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/Shaders" "$<TARGET_FILE_DIR:DXFramework>/Shaders"
        COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/Assets" "$<TARGET_FILE_DIR:DXFramework>/Assets")
endif()
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(Tests)
endif()
